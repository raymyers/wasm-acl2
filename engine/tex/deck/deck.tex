\begin{code}{\codesmall\baselineskip8pt}
rule \red{Step_pure/if-true}:
  (CONST I32 c) (\hi{IF} bt instr_1* ELSE instr_2*)  ~>  (BLOCK bt instr_1*)
  -- if \hi{c =/= 0}
\end{code}
  \end{minipage}\par
  \vspace*{5mm}
  \noindent\begin{minipage}{.25\textwidth}
    \begin{center}\LARGE ACL2 rule\\\muted{\large (cleaned up)}\end{center}
  \end{minipage}\hfill
  \begin{minipage}{.73\textwidth}
\begin{code}{\codenine\baselineskip11pt}
(defun if-true (instrs)
  (case-match instrs
    (((:const (:i32) c) (\hi{:ifelse} bt then else))
     (and (true-listp then)
          (true-listp else)
          \hi{(not (equal c 0))}
          (list (list (list :block bt then)))))))
\end{code}
  \end{minipage}\par
  \vfill
\end{slide}

\begin{slide}

  \begin{minipage}{.73\textwidth}
\begin{code}{\codesmall\baselineskip8pt}
rule \red{Step_pure/if-true}:
  (CONST I32 c) (\hi{IF} bt instr_1* ELSE instr_2*)  ~>  (BLOCK bt instr_1*)
  -- if \hi{c =/= 0}
\end{code}
    \vspace*{2mm}
\begin{code}{\codetiny\baselineskip5.2pt}
(AND (TRUE-LISTP %A0)
     (EQUAL (ACL2::LEN %A0) 2)
     (LET* ((%T16 (CAR %A0)))
       (AND (CONSP %T16)
            (EQ (CAR %T16) :CONST)
            (TRUE-LISTP (CDR %T16))
            (EQUAL (ACL2::LEN (CDR %T16)) 2)
            (LET* ((%T17 (CAR (CDR %T16))))
              (AND (CONSP %T17)
                   (EQ (CAR %T17) :I32)
                   (TRUE-LISTP (CDR %T17))
                   (EQUAL (ACL2::LEN (CDR %T17)) 0)
                   (LET* ((C (CAR (CDR (CDR %T16))))
                          (%T18 (CAR (CDR %A0))))
                     (AND (CONSP %T18)
                          (EQ (CAR %T18) \hi{:IFELSE})
                          (TRUE-LISTP (CDR %T18))
                          (EQUAL (ACL2::LEN (CDR %T18)) 3)
                          (LET* ((BT (CAR (CDR %T18)))
                                 (%T19 (CAR (CDR (CDR %T18)))))
                            (AND (TRUE-LISTP %T19)
                                 (LET* ((INSTR_1-STAR %T19)
                                        (%T20 (CAR (CDR (CDR (CDR %T18))))))
                                   (AND (TRUE-LISTP %T20)
                                        (LET* ((INSTR_2-STAR %T20))
                                          (AND \hi{(NOT (EQUAL C 0))}
                                               (LIST (%RULE "Step_pure/if-true"
                                                            (LIST (LIST :BLOCK
                                                                        BT
                                                                        INSTR_1-STAR)))))))))))))))))
\end{code}
